\begin{theorem}[Rough squarefree-modulus variance]\label{thm:roughvariance-main}
Let $R\ge2$, $D\ge1$, $P\ge2$, and let $j$ be fixed.  Let $\cA$ be any set of
squarefree integers $q\in[R,2R)$ satisfying
\[
       \omega(q)\le j,\qquad P^-(q)\ge P.
\]
Let $(a_n)$ be a \emph{single common} sequence with $|a_n|\le1$, supported on
$D<n\le2D$.  For $D\le y\le2D$ and $(a,q)=1$, put
\[
 E_q(a;y)=
 \sum_{\substack{D<n\le y\\n\equiv a\pmod q}}a_n
 -\frac1{\phi(q)}\sum_{\substack{D<n\le y\\(n,q)=1}}a_n.
\]
Then, uniformly in $y$ and in $(a_n)$,
\begin{equation}\label{eq:roughvariance-main}
 \sum_{q\in\cA}\sum_{a\bmod q}^{*}|E_q(a;y)|^2
       \ll_j DR+\frac{D^2}{P}.
\end{equation}
Under the standard primitive-large-sieve normalisation used in
\cref{sec:roughproof}, the conservative coefficient $2^{2j+8}$ is admissible
up to endpoint rounding.
\end{theorem}

\subsection{Exact combinatorial normalisation}

\classification{\newreduction}{Kernel checked in \path{Gate1B/R11/Card5.lean} and \texttt{Factor542.lean}.}
\begin{proposition}[CARD5 and $4\mid4\mid2$ ledger]\label{prop:normalisation}
The CARD5 selector chooses five of ten large coordinates and has cardinality
\[
                  \binom{10}{5}=252.
\]
Together with the distinguished prime, every selected divisor has six distinct
prime factors and therefore M\"obius sign $(-1)^6=+1$.  The ordered
$4\mid4\mid2$ allocations have cardinality
\[
  \binom{10}{4}\binom{6}{4}=3150
   =\frac{10!}{4!\,4!\,2!}.
\]
The factors $1/4!$, $1/4!$, $1/2!$, and $1/3150$ consume the complete internal
multiplicity.  Consequently, the outer CARD5 coefficient is $252$, not
$252\cdot3150$, and
\[
                    \frac{252}{3150}=\frac2{25}.
\]
\end{proposition}

\begin{proof}
The cardinalities are the displayed binomial coefficients.  For a
representation-blind atom weight, the three grouped products in any
$4\mid4\mid2$ allocation recombine to the full ten-atom product.  Summing over
all $3150$ allocations therefore gives $3150$ identical copies, which are
cancelled by the averaging factor.  The remaining factor is the CARD5 count.
The formal proof additionally checks distinctness, squarefreeness, the
M\"obius sign, and the factorial identity in the kernel.
\end{proof}

\begin{warning}[Representation-blindness]
The equal-$n$ CARD5 collapse is formalised only for a weight whose type excludes
dependence on the selected five-subset.  A state-dependent label cannot be
silently collapsed to $252$ copies.
\end{warning}
